function ignorarCartaoRegra($dezenas, $quantidade, $cartao) {

    $dezenasRegra = explode(',', $dezenas);
    $quantidadeRegra = 0;

    foreach ($cartao as $dez) {
        if(in_array($dez, $dezenasRegra)) {
            $quantidadeRegra++;
        }
    }

    return $quantidadeRegra != $quantidade;
}

function ignorarCartaoRegraAte($dezenas, $quantidade, $cartao) {

    $dezenasRegra = explode(',', $dezenas);
    $quantidadeRegra = 0;

    foreach ($cartao as $dez) {
        if(in_array($dez, $dezenasRegra)) {
            $quantidadeRegra++;
        }
    }

    return $quantidadeRegra > $quantidade;
}

function gerarDezenasCombinadas($dezenaMaxima, $dezenasRemover)
{
    return array_diff(range(1, $dezenaMaxima), $dezenasRemover);
}

function tratarPares($quantidadePares, $loteria, $cartoes) 
{
    foreach ($cartoes as $cartao => $linha) {
        $dezenas = explode(',', $linha);

        $pares = 0;
        $impares = 0;

        foreach ($dezenas as $dezena) {
            if ($dezena % 2 == 0) {
                $pares++;
                continue;
            } 

            $impares++;
        }

        if ($quantidadePares != $pares) {
            unset($cartoes[$cartao]);
        }
    }

    return $cartoes;

}
